--- PetService.java ---
public class PetService {
    private final PetRepository repo = new PetRepository();
    public void save(PetEntity pet) { repo.save(pet); }
    public List<PetEntity> findAll() { return repo.findAll(); }
}

--- SavePetServlet.java ---
@WebServlet("/save-pet")
public class SavePetServlet extends HttpServlet {
    private final PetService service = new PetService();
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws IOException {
        req.setCharacterEncoding("UTF-8");
        PetEntity pet = new PetEntity();
        pet.setPetName(req.getParameter("petName"));
        service.save(pet);
        resp.sendRedirect("list.jsp");
    }
}

--- SoapPetDataService.java ---
@WebService(serviceName = "PetDataService")
public class SoapPetDataService {
    private final PetRepository repo = new PetRepository();
    @WebMethod(operationName = "getAllPets")
    public List<PetEntity> getAllPets() { return repo.findAll(); }
}
