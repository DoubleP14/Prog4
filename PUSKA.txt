3. Lépés: Repository Réteg (JNDI és JDBC)
Minden adatbázisos feladathoz kell. Ez egy univerzális blokk a JNDI lekéréshez és egy alap SELECT illetve INSERT művelethez.

Java
import javax.naming.Context;
import javax.naming.InitialContext;
import javax.sql.DataSource;
import java.sql.*;
import java.util.ArrayList;
import java.util.List;

public class UniversalRepository {
    private DataSource dataSource;

    public UniversalRepository() {
        try {
            Context initCtx = new InitialContext();
            Context envCtx = (Context) initCtx.lookup("java:comp/env");
            // A névnek egyeznie kell a context.xml-ben lévővel!
            this.dataSource = (DataSource) envCtx.lookup("jdbc/ZH_MariaDB");
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // LISTÁZÁS (READ)
    public List<MyEntity> getAll() {
        List<MyEntity> list = new ArrayList<>();
        String sql = "SELECT * FROM tabla_nev";
        
        try (Connection conn = dataSource.getConnection();
             PreparedStatement ps = conn.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
             
            while (rs.next()) {
                MyEntity entity = new MyEntity();
                entity.setId(rs.getLong("id"));
                entity.setName(rs.getString("name"));
                list.add(entity);
            }
        } catch (SQLException e) {
            e.printStackTrace();
        }
        return list;
    }

    // MENTÉS (CREATE)
    public void save(MyEntity entity) {
        String sql = "INSERT INTO tabla_nev (name, value) VALUES (?, ?)";
        
        try (Connection conn = dataSource.getConnection();
             PreparedStatement ps = conn.prepareStatement(sql)) {
             
            ps.setString(1, entity.getName());
            ps.setInt(2, entity.getValue());
            ps.executeUpdate();
            
        } catch (SQLException e) {
            e.printStackTrace();
        }
    }
}

5. Lépés: REST Endpoint hívása JavaScriptből (Frontend)
Ez a "Gomb implementálása" feladatokhoz kell, ami a böngészőből dobja fel az OK / Hiba történt ablakokat.

JavaScript
// Ezt a JSP oldal aljára a <script> tagek közé tedd!
function sendRestRequest(entityId) {
    // A pontos URL a REST controlleredtől függ!
    const url = 'http://localhost:8080/api/endpoint/' + entityId;

    fetch(url, {
        method: 'GET', // Vagy POST, feladattól függ
        headers: {
            'Content-Type': 'application/json'
        }
    })
    .then(response => {
        if (!response.ok) {
            throw new Error('Network response was not ok');
        }
        return response.json();
    })
    .then(data => {
        // Sikeres válasz esetén alert:
        alert("Sikeres hívás: " + JSON.stringify(data));
    })
    .catch(error => {
        // Hiba esetén alert:
        alert("Hiba történt!");
        console.error('There has been a problem with your fetch operation:', error);
    });
}

Amikor generáltatsz kliens osztályokat WSDL-ből, utána a te Service rétegedben így kell meghívni:

public class MyService {
    // A pontos osztálynevek a WSDL-ből generált kódtól függnek!
    public String callSoapService(Long objectId) {
        try {
            // 1. Példányosítsd a Service osztályt
            MyGeneratedWebService service = new MyGeneratedWebService();
            
            // 2. Kérd el a portot
            MyGeneratedPortType port = service.getMyGeneratedPort();
            
            // 3. Hívd meg a metódust
            String result = port.getSomeData(objectId);
            return result;
            
        } catch (Exception e) {
            e.printStackTrace();
            return null;
        }
    }
}
