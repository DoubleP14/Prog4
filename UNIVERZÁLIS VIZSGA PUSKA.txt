--- 4. JAX-RS (REST API Controller) ---

import jakarta.ws.rs.*;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;

@Path("/api/data")
@Produces(MediaType.APPLICATION_JSON)
@Consumes(MediaType.APPLICATION_JSON)
public class MyRestController {

    private final MyService service = new MyService();

    @GET
    @Path("/{id}")
    public Response getById(@PathParam("id") Long id) {
        MyEntity data = service.findById(id);
        if (data != null) {
            return Response.ok(data).build();
        }
        return Response.status(Response.Status.NOT_FOUND).entity("{\"error\":\"Nem talalhato\"}").build();
    }

    @POST
    public Response create(MyEntity entity) {
        service.save(entity);
        return Response.status(Response.Status.CREATED).entity(entity).build();
    }
}


--- 5. JAX-WS (SOAP Kliens hívása dinamikus URL-lel) ---

import jakarta.xml.ws.BindingProvider;

public String callRemoteWebService(String id) {
    MyRemoteService_Service service = new MyRemoteService_Service();
    MyRemoteService port = service.getMyRemoteServicePort();

    String valosUrl = "http://localhost:8081/ws-app/ws/remote-data";
    ((BindingProvider) port).getRequestContext()
            .put(BindingProvider.ENDPOINT_ADDRESS_PROPERTY, valosUrl);

    return port.getRemoteData(id);
}


--- 6. SERVLET: A "PRG" Mentés (Post-Redirect-Get) ---

@WebServlet(urlPatterns = "/save-data")
public class SaveServlet extends HttpServlet {
    private final MyService service = new MyService();

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws IOException {
        req.setCharacterEncoding("UTF-8");
        
        try {
            String name = req.getParameter("name");
            service.save(new MyEntity(name));
            resp.sendRedirect(req.getContextPath() + "/list");
        } catch (Exception e) {
            req.setAttribute("hiba", "Rossz adat!");
            req.getRequestDispatcher("/form.jsp").forward(req, resp);
        }
    }
}
